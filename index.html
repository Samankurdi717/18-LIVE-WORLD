const APP_ID = "3b947b82132840c6942bbfa2045eac18"; let client = null; let localAudio = null; let localVideo = null; let cameraOn = true; let micOn = true; async function joinCall() { const age = document.getElementById("age"); const channelInput = document.getElementById("channel"); const status = document.getElementById("status"); if (!age || !channelInput || !status) { alert("هەڵە: بەشەکانی پەڕە نەدۆزرایەوە."); return; } if (!age.checked) { alert("پێویستە 18 ساڵ یان زیاتر بیت."); return; } if (client) { alert("پێشتر بەشداری چەناڵت کردووە."); return; } const channel = channelInput.value.trim(); if (!channel) { alert("ناوی چەناڵ بنووسە."); return; } try { status.innerText = "چاوەڕوان بە..."; if (typeof AgoraRTC === "undefined") { throw new Error("Agora SDK بار نەبوو."); } client = AgoraRTC.createClient({ mode: "rtc", codec: "vp8" }); client.on("user-published", async (user, mediaType) => { await client.subscribe(user, mediaType); if (mediaType === "video") { let div = document.getElementById("remote-" + user.uid); if (!div) { div = document.createElement("div"); div.id = "remote-" + user.uid; document.getElementById("remote").appendChild(div); } user.videoTrack.play(div); } if (mediaType === "audio") { user.audioTrack.play(); } }); client.on("user-left", (user) => { const div = document.getElementById("remote-" + user.uid); if (div) div.remove(); }); const uid = Math.floor(Math.random() * 100000) + 1; const response = await fetch( `https://18-live-world.vercel.app/token?channel=${encodeURIComponent(channel)}&uid=${uid}` ); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "Token error"); } await client.join(APP_ID, channel, data.token, uid); [localAudio, localVideo] = await AgoraRTC.createMicrophoneAndCameraTracks(); localVideo.play("local"); await client.publish([localAudio, localVideo]); status.innerText = "بە سەرکەوتوویی چوویتە ناو چەناڵ"; } catch (e) { console.error("Join error:", e); await leaveCall(); status.innerText = "هەڵە: " + e.message; alert("هەڵە: " + e.message); } } async function leaveCall() { try { if (localAudio) { localAudio.close(); localAudio = null; } if (localVideo) { localVideo.close(); localVideo = null; } if (client) { await client.leave(); client = null; } document.getElementById("local").innerHTML = ""; document.getElementById("remote").innerHTML = ""; document.getElementById("status").innerText = "دەرچوویت"; } catch (e) { console.error("Leave error:", e); } } async function toggleCamera() { if (localVideo) { cameraOn = !cameraOn; await localVideo.setEnabled(cameraOn); } } async function toggleMic() { if (localAudio) { micOn = !micOn; await localAudio.setEnabled(micOn); } }
